132 void applyStep(Manager& manager, int transitionToStep)
133 {
134 // Nodes
135
136 auto stepIt = manager.steps.begin();
137 std::advance(stepIt, transitionToStep);
138
139 for (auto& node : stepIt->nodes) {
140 NodeEntity* t_node = node.first;
141
142 if (t_node->hasComponent<TransformComponent>() &&
143 //t_node->hasComponent<MovingAnimatorComponent>() &&
144 t_node->hasComponent<RectangleFlashAnimatorComponent>()
145 ) {
146
147 auto& tc = t_node->GetComponent<TransformComponent>();
148
149 /*t_node->GetComponent<MovingAnimatorComponent>().Play("StepMove",*/
150 tc.position = node.second.position;
151
152 tc.size = glm::vec3(node.second.size);
153
154 t_node->GetComponent<RectangleFlashAnimatorComponent>().Play("RectInterpolation", node.second.color);
155
156 }
157 }
158
159 // Links
160 for (auto& link : stepIt->links) {
161 LinkEntity* t_link = link.first;
162
163 if (t_link->hasComponent<Line_w_Color>()) {
164 auto& lwc = t_link->GetComponent<Line_w_Color>();
165 lwc.src_color = link.second.color;
166 lwc.dest_color = link.second.color;
167 lwc.width = link.second.width;
168 }
169 }
170
171 for (auto& simPaths : DataManager::getInstance().mapSimToGraphPaths) {
172 auto& pathLinks = simPaths.second->GetComponent<PathLinkerComponent>().pathLinks;
173 auto toRemove = pathLinks;
174
175 for (auto pathLinkId : toRemove) {
176 simPaths.second->GetComponent<PathLinkerComponent>().removeLink(pathLinkId);
177 Entity* pathLink = manager.getEntityFromId(pathLinkId);
178 pathLink->destroy();
179 }
180 }
181
182 // Paths
183 for (size_t i = 0; i < stepIt->paths.size(); i++) {
184
185 auto stepIt = manager.steps.begin();
186 std::advance(stepIt, transitionToStep);
187
188 auto& path = stepIt->paths[i];
189
190 auto& plc = path.first->GetComponent<PathLinkerComponent>();
191
192 plc.color = path.second.color;
193 plc.width = path.second.width;
194
195 for (auto linkId : path.second.link_ids) {
196 // Find the link entity by ID
197 LinkEntity* linkEntity = nullptr;
198 for (auto& link : DataManager::getInstance().mapSimToGraphLinks) {
199 if (link.second->getId() == linkId) {
200 linkEntity = link.second;
201 break;
202 }
203 }
204
205 if (!linkEntity) break;
206
207 int idA = std::get<int>(linkEntity->getFromNode());
208 int idB = std::get<int>(linkEntity->getToNode());
209
210 // create ECS link
211 auto& link = manager.addEntity<Link>((int)idA, (int)idB);
212 link.addToGroup(Manager::groupPathLinks);
213 link.addComponent<Line_w_Color>();
214
215 link.addComponent<LineFlashAnimatorComponent>();
216
217 manager.grid->addLink(&link, manager.grid->getGridLevel());
218
219 // associate link with path linker
220 path.first->GetComponent<PathLinkerComponent>().addLink(link.getId());
221
222 }
223 }
224
225 // PATH LINKS UPDATE
226 if (manager.arrowheadsEnabled) {
227 for (auto* link : manager.getGroup<LinkEntity>(Manager::groupPathLinks))
228 {
229 auto* link_entity = dynamic_cast<LinkEntity*>(link);
230
231 link_entity->setConnectionType(LinkEntity::ConnectionType::PORT_TO_PORT);
232 link_entity->updateConnection();
233 }
234
235 for (auto* pathLinker : manager.getGroup<EmptyEntity>(Manager::groupPathLinksHolder))
236 {
237 pathLinker->GetComponent<PathLinkerComponent>().createInnerLinks();
238 }
239 }

324 std::string simulationStepToString(const SimulationStep& step)
325 {
326 std::ostringstream out;
327
328 out << "=== Simulation Step ===\n";
329 out << "Index: " << step.step_index << "\n";
330 out << "Timestamp: " << step.timestamp << "\n";
331
332 // ---- Nodes ----
333 out << "Nodes (" << step.nodes.size() << ")\n";
334 for (const auto& [node, simNode] : step.nodes) {
335 out << " Node ptr: " << node
336 << " pos: (" << simNode.position.x
337 << ", " << simNode.position.y
338 << ", " << simNode.position.z << ")"
339 << " size: " << simNode.size.value
340 << "\n";
341 }
342
343 // ---- Links ----
344 out << "Links (" << step.links.size() << ")\n";
345 for (const auto& [link, simLink] : step.links) {
346 out << " Link ptr: " << link
347 << " width: " << simLink.width
348 << "\n";
349 }
350
351 // ---- Paths ----
352 out << "Paths (" << step.paths.size() << ")\n";
353 for (const auto& [pathEntity, simPath] : step.paths) {
354 out << " Path ptr: " << pathEntity
355 << " width: " << simPath.width
356 << " links: ";
357
358 for (EntityID id : simPath.link_ids) {
359 out << EntityIDUtils::toString(id) << " ";
360 }
361 out << "\n";
362 }
363
364 out << "=======================\n";
365
366 return out.str();
367 }
